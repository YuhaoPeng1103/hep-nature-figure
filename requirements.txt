# hep-nature-figure 依赖
#
# ★ 生成 SVG 本身是零依赖的（纯字符串拼接）。
#   下面这些是"渲染 / 审计 / 度量"需要的。
#
# 实测环境：ChatGPT Code Interpreter 自带 numpy/matplotlib/scipy/pandas/
# Pillow/PyMuPDF/CairoSVG；**没有**网络，装不了包。

# ── 核心（几乎所有工具都要）──
numpy>=1.24
Pillow>=10.0
cairosvg>=2.7          # SVG → PNG/PDF 渲染。★必须用它，PyMuPDF 渲 SVG 会丢渐变
pymupdf>=1.24          # PDF 几何审计、拼版、从论文切图

# ── 位图 → 矢量（scripts/raster_to_vector_semantic.py）──
#   numpy + scipy + Pillow + cairosvg + cairocffi + fonttools
#   ★ 不需要 cv2 / skimage —— 那是下面逐像素描摹（raster_to_vector.py）才要的
cairocffi>=1.6         # labels.py 栅格化文字做逐词对齐（★必须与成品同渲染器）
fonttools>=4.40        # labels.py 读字体度量表 —— 量字宽的字体必须与 SVG 写的族名一致

# ── 按用途选装 ──
matplotlib>=3.7        # 定量图（T1/T2）
scipy>=1.10            # 统计、插值；raster_vector 的连通域/元素切分也用它
shapely>=2.0           # geom.py 的布尔运算（无它时 blob()/poly_to_path() 仍可用）
PyYAML>=6.0            # 读 IR 的 yaml。无它时门禁自动跳过结构断言，不阻断
opencv-python>=4.8     # raster_to_vector.py 逐像素描摹（无 skimage 时用它）
scikit-image>=0.21     # raster_to_vector.py 逐像素描摹的备选后端（二选一即可）
